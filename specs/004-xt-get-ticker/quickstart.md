## Python API 测试

```python
import asyncio
from tri_arb.arbitrage import ArbitrageMonitor
from tri_arb.arbitrage.config import MonitorConfig

async def test_scan_once():
    config = MonitorConfig(min_profit_threshold=0.5)
    monitor = ArbitrageMonitor(config=config, exchange_name="xt")

    opportunities = await monitor.scan_once()

    print(f"Found {len(opportunities)} opportunities")
    for opp in opportunities:
        print(f"  {opp.path.trading_pairs}: {opp.expected_profit_rate}%")
        assert opp.expected_profit_rate >= 0.5  # 验证阈值

asyncio.run(test_scan_once())
```
